#cmake version 
cmake_minimum_required(VERSION 3.16)

# PROJECT
project(TicTacToe VERSION 1.0.0)

# C++
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# CMake
include(FetchContent)

# download raylib from github
FetchContent_Declare(
    raylib
    GIT_REPOSITORY https://github.com/raysan5/raylib.git
    GIT_TAG        5.5  # Pulls the stable 5.5 release safely
)

# turn off examples and games 
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_GAMES OFF CACHE BOOL "" FORCE)

# prepare Raylib right now
FetchContent_MakeAvailable(raylib)

#target include directories
include_directories(${CMAKE_SOURCE_DIR}/include)

# starting point-> Main.cpp file
add_executable(TicTacToe src/Main.cpp src/render.cpp src/game.cpp)


# target link libraries 
target_link_libraries(TicTacToe PRIVATE raylib)
